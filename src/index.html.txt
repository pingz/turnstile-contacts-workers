    <script>
      function onSuccess(token) {
	  // 提交表单到 Worker
	  fetch('/verify', {
              method: 'POST',
              body: new URLSearchParams({ 'cf-turnstile-response': token })
	  })
	      .then(response => {
		  if (response.ok) return response.text();
		  throw new Error('Verification failed');
	      })
	      .then(html => {
		  // 用邮箱信息替换验证区域
		  document.querySelector('.cf-turnstile').closest('form').remove();
		  document.getElementById('email-container').innerHTML = html;
	      })
	      .catch(error => {
		  console.error('Error:', error);
		  alert('校验失败. 请重试.');
		  turnstile.reset(); // 重置验证
	      });
      }
    </script>
